import { createRequire } from 'node:module';
const require = createRequire('/home/user/qf/apps/web/package.json');
const { chromium } = require('playwright-core');
const BASE = 'http://127.0.0.1:8788';
const OUT = '/tmp/claude-0/m7/shots/';
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium', args: ['--no-sandbox'] });
const problems = [];
for (const [name, viewport] of [['live-earnings-desktop', { width: 1280, height: 900 }], ['live-earnings-phone', { width: 390, height: 844 }]]) {
  const ctx = await browser.newContext({ viewport, colorScheme: 'dark', deviceScaleFactor: 1 });
  const page = await ctx.newPage();
  page.on('console', (m) => { if (m.type() === 'error') problems.push(`${name} console.error: ${m.text()}`); });
  page.on('pageerror', (e) => problems.push(`${name} pageerror: ${e.message}`));
  await page.goto(`${BASE}/#/earnings`);
  await page.waitForSelector('.card.treasury', { timeout: 60000 });
  await page.waitForTimeout(800);
  await page.screenshot({ path: `${OUT}${name}.png`, fullPage: true });
  await page.locator('.card.treasury').screenshot({ path: `${OUT}${name}-treasury-card.png` });
  await ctx.close();
}
await browser.close();
console.log(JSON.stringify({ problems }));
